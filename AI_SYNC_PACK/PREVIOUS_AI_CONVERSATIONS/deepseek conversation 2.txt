from bleak import BleakClient

STRIP_MAC = "41:42:43:E7:8B:F6" 
# The correct UUID for writing commands (from your Bleak.py output)
CHAR_UUID = "0000fff3-0000-1000-8000-00805f9b34fb"

# Known MR Star commands (from the reverse-engineering of GATT--DEMO)
CMD_ON = "bc01010001"  # Turn ON (sometimes bc01010000 is off, let's test)
CMD_OFF = "bc01010055" # Turn OFF

# Wait, the known command for red was bc0406000003e8000055.

# Note: MR Star uses a specific 4-byte color format. 
# For a simple test, let's use the exact Red command found online:
CMD_RED = "bc0406000003e8000055" 

async def main():
    print(f"Connecting to {STRIP_MAC}...")
    client = BleakClient(STRIP_MAC)
    
    try:
        await client.connect()
        print("Connected!")
        
        # Helper function to send commands safely
        async def send_cmd(name, hex_str):
            if not client.is_connected:
                print(f"[{name}] Disconnected! Reconnecting...")
                await client.connect()
            
            print(f"Sending {name}...")
            # Convert hex string to bytes
            data = bytes.fromhex(hex_str)
            await client.write_gatt_char(CHAR_UUID, data, response=False)
            print(f"{name} sent.")
            await asyncio.sleep(1) # Give the strip time to process

        # Test the commands
        print("\n--- Testing MR Star Commands ---")
        
        # 1. Try to turn it OFF
        await send_cmd("OFF", CMD_OFF)
        
        # 2. Try to turn it ON
        await send_cmd("ON", CMD_ON)
        
        # 3. Try to set it to RED
        await send_cmd("RED", CMD_RED)
        
        print("\nDone! Did the strip react?")
        
    except Exception as e:
        print(f"Error: {e}")
    finally:
        if client.is_connected:
            await client.disconnect()
            print("Disconnected.")

if __name__ == "__main__":
    asyncio.run(main())

import asyncio

async def main():

    client = BleakClient(STRIP_MAC)
    
    try:
        await client.connect()
        print("Connected!")
        await asyncio.sleep(1) # Give it a moment to settle

            await client.write_gatt_char(CHAR_UUID, data, response=False)
            print(f"{name} sent.")
            await asyncio.sleep(1.5) # Give the strip time to process

    except Exception as e:
        print(f"Error: {e}")
    finally:
        if client.is_connected:
