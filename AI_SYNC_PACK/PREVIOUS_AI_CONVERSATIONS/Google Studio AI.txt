#!/usr/bin/env python3
"""
AymashTain LED RGB Remote - Python Native Hardware Controller & Verification Suite
Directly controls MR Star, Triones, and HappyLighting BLE LED strips via Windows/Linux Bluetooth.

Requirements:
    pip install bleak opencv-python numpy
"""

import sys
import time
import asyncio
import colorsys
from typing import Optional, List, Dict

try:
    from bleak import BleakClient, BleakScanner
except ImportError:
    print("\n[ERROR] 'bleak' library is not installed.")
    print("Run: pip install bleak opencv-python numpy\n")
    sys.exit(1)

try:
    import cv2
    import numpy as np
    HAS_OPENCV = True
except ImportError:
    HAS_OPENCV = False

# ==============================================================================
# Hardware Configurations & Protocol Constants
# ==============================================================================
KNOWN_DEVICES = {
    "1": {"name": "Strip 1 (Desk Left)",   "address": "41:42:59:F1:C8:68"},
    "2": {"name": "Strip 2 (Desk Center)", "address": "41:42:43:E7:8B:F6"},
    "3": {"name": "Strip 3 (Wall Ambient)","address": "41:42:F9:D7:45:B0"},
}

CANDIDATE_WRITE_UUIDS = [
    "00002022-0000-1000-8000-00805f9b34fb", # MR Star custom service
    "0000fff3-0000-1000-8000-00805f9b34fb", # Triones / HappyLighting write char
    "0000ffe1-0000-1000-8000-00805f9b34fb", # CC2541 / HM-10 transparent UART
    "0000ffd1-0000-1000-8000-00805f9b34fb",
    "0000ffa1-0000-1000-8000-00805f9b34fb",
]

# ==============================================================================
# Protocol Encoding Helpers
# ==============================================================================
def encode_mrstar_power(power_on: bool) -> bytes:
    """BC 01 01 01 55 (ON) or BC 01 01 00 55 (OFF)"""
    return bytes([0xBC, 0x01, 0x01, 0x01 if power_on else 0x00, 0x55])

def encode_mrstar_color(hue: int, sat: int = 1000) -> bytes:
    """BC 04 06 <H_hi> <H_lo> <S_hi> <S_lo> 00 00 55"""
    h = max(0, min(360, int(hue)))
    s = max(0, min(1000, int(sat)))
    return bytes([
        0xBC, 0x04, 0x06,
        (h >> 8) & 0xFF, h & 0xFF,
        (s >> 8) & 0xFF, s & 0xFF,
        0x00, 0x00, 0x55
    ])

def encode_mrstar_brightness(brightness: int = 1000) -> bytes:
    """BC 05 06 <B_hi> <B_lo> <B_hi> <B_lo> 00 00 55"""
    b = max(0, min(1000, int(brightness)))
    return bytes([
        0xBC, 0x05, 0x06,
        (b >> 8) & 0xFF, b & 0xFF,
        (b >> 8) & 0xFF, b & 0xFF,
        0x00, 0x00, 0x55
    ])

def encode_triones_rgb(r: int, g: int, b: int) -> bytes:
    """7E 07 05 03 R G B 00 EF"""
    return bytes([0x7E, 0x07, 0x05, 0x03, r & 0xFF, g & 0xFF, b & 0xFF, 0x00, 0xEF])

def rgb_to_hsv(r: int, g: int, b: int):
    h, s, v = colorsys.rgb_to_hsv(r / 255.0, g / 255.0, b / 255.0)
    return int(h * 360), int(s * 1000), int(v * 1000)

def timestamp_str() -> str:
    return time.strftime("%H:%M:%S", time.localtime())

# ==============================================================================
# Controller Class
# ==============================================================================
class BleLedController:
    def __init__(self):
        self.client: Optional[BleakClient] = None
        self.write_char_uuid: Optional[str] = None
        self.connected_address: Optional[str] = None
        self.connected_name: Optional[str] = None

    async def scan(self, timeout: float = 5.0) -> List[Dict[str, str]]:
        print(f"[*] Scanning for nearby BLE devices ({timeout}s)...")
        devices = await BleakScanner.discover(timeout=timeout)
        found = []
        for d in devices:
            name = d.name or "Unknown"
            found.append({"name": name, "address": d.address})
        return found

    async def connect(self, address: str, name: str = "LED Strip") -> bool:
        print(f"[*] Connecting to {name} [{address}]...")
        try:
            self.client = BleakClient(address)
            await self.client.connect()
            self.connected_address = address
            self.connected_name = name
            print(f"[+] Connected to {address} successfully!")

            # Auto-detect writable GATT characteristic
            self.write_char_uuid = None
            for service in self.client.services:
                for char in service.characteristics:
                    props = char.properties
                    if "write" in props or "write-without-response" in props:
                        if char.uuid.lower() in [u.lower() for u in CANDIDATE_WRITE_UUIDS]:
                            self.write_char_uuid = char.uuid
                            break
                        if not self.write_char_uuid:
                            self.write_char_uuid = char.uuid

            if self.write_char_uuid:
                print(f"[+] Using write characteristic: {self.write_char_uuid}")
            else:
                self.write_char_uuid = CANDIDATE_WRITE_UUIDS[0]

            # Physical connection blink confirmation
            await self.blink_strip()
            return True

        except Exception as e:
            print(f"[-] Connection failed: {e}")
            self.client = None
            return False

    async def disconnect(self):
        if self.client and self.client.is_connected:
            await self.client.disconnect()
            print("[*] Disconnected.")
        self.client = None

    async def send_raw(self, data: bytes, response: bool = False) -> bool:
        if not self.client or not self.client.is_connected:
            print("[-] Error: Not connected to any BLE device.")
            return False
        try:
            await self.client.write_gatt_char(self.write_char_uuid, data, response=response)
            return True
        except Exception:
            try:
                await self.client.write_gatt_char(self.write_char_uuid, data, response=not response)
                return True
            except Exception as e:
                print(f"[-] Write error: {e}")
                return False

    async def set_power(self, on: bool) -> bool:
        pkt = encode_mrstar_power(on)
        ok = await self.send_raw(pkt)
        status = "ON" if on else "OFF"
        print(f"[{timestamp_str()}] Power {status}: {pkt.hex().upper()} -> {'OK' if ok else 'FAIL'}")
        return ok

    async def set_color_hsv(self, hue: int, sat: int = 1000, brightness: int = 1000) -> bool:
        col_pkt = encode_mrstar_color(hue, sat)
        ok1 = await self.send_raw(col_pkt)
        await asyncio.sleep(0.04)
        bri_pkt = encode_mrstar_brightness(brightness)
        ok2 = await self.send_raw(bri_pkt)
        print(f"[{timestamp_str()}] Color (H:{hue}, S:{sat}): {col_pkt.hex().upper()} -> {'OK' if ok1 and ok2 else 'FAIL'}")
        return ok1 and ok2

    async def set_color_rgb(self, r: int, g: int, b: int) -> bool:
        h, s, v = rgb_to_hsv(r, g, b)
        ok = await self.set_color_hsv(h, s, v)
        await self.send_raw(encode_triones_rgb(r, g, b))
        return ok

    async def blink_strip(self):
        """Blink confirmation indicator on physical strip"""
        print("[*] Triggering connection blink indicator on physical strip...")
        await self.set_power(False)
        await asyncio.sleep(0.18)
        await self.set_power(True)
        await asyncio.sleep(0.12)
        await self.set_color_hsv(180, 1000, 1000) # Cyan flash
        await asyncio.sleep(0.15)
        await self.set_color_hsv(0, 1000, 1000)   # Red default

    async def run_sweep(self, steps: int = 12, gap_ms: int = 400):
        """
        Run the 12-command hue sweep:
        0°, 30°, 60°, 90°, 120°, 150°, 180°, 210°, 240°, 270°, 300°, 330°
        """
        gap_sec = max(0.05, gap_ms / 1000.0)
        print(f"\n[Sweep] Starting sequence with {steps} commands. Gap: {gap_ms}ms...")
        step_angle = 360 // steps

        for idx in range(steps):
            hue = (idx * step_angle) % 360
            pkt = encode_mrstar_color(hue, 1000)
            hex_cmd = pkt.hex().upper()
            ok = await self.send_raw(pkt)
            print(f"[{idx + 1}/{steps}] {timestamp_str()} {hex_cmd} -> {'OK' if ok else 'FAIL'}")
            await asyncio.sleep(gap_sec)

        print("[Sweep] Completed sweep run.\n")

# ==============================================================================
# OpenCV Camera Color Verification
# ==============================================================================
def run_camera_verification(controller: BleLedController):
    if not HAS_OPENCV:
        print("\n[-] OpenCV (cv2) is not installed. Run: pip install opencv-python numpy\n")
        return

    print("\n[*] Opening camera for real-time optical color verification...")
    cap = cv2.VideoCapture(0)
    if not cap.isOpened():
        print("[-] Error: Could not open webcam.")
        return

    cv2.namedWindow("AymashTain LED Remote - Camera Verification", cv2.WINDOW_NORMAL)
    cv2.resizeWindow("AymashTain LED Remote - Camera Verification", 800, 600)
    loop = asyncio.get_event_loop()
    target_rgb = [255, 0, 0]

    while True:
        ret, frame = cap.read()
        if not ret:
            break

        h, w, _ = frame.shape
        rw, rh = int(w * 0.4), int(h * 0.3)
        rx, ry = (w - rw) // 2, (h - rh) // 2

        roi = frame[ry:ry + rh, rx:rx + rw]
        avg_bgr = np.mean(roi, axis=(0, 1))
        meas_b, meas_g, meas_r = int(avg_bgr[0]), int(avg_bgr[1]), int(avg_bgr[2])

        cv2.rectangle(frame, (rx, ry), (rx + rw, ry + rh), (0, 255, 0), 2)
        cv2.putText(frame, "Optical ROI Target", (rx + 5, ry - 8),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1, cv2.LINE_AA)

        cv2.rectangle(frame, (10, 10), (450, 95), (20, 20, 25), -1)
        cv2.putText(frame, f"Measured RGB: ({meas_r}, {meas_g}, {meas_b})",

            fields = [
                "timestamp", "step_id", "step_label", "command_label", "hex_command",
                "address", "characteristic", "accepted", "duration_ms", "retry_used",
                "connected_before", "connected_after", "error"
            ]
            writer = csv.DictWriter(file, fieldnames=fields)
            writer.writeheader()
            for r in self.device_results:
                writer.writerow(asdict(r))
        self.event(f"Saved log: {self.text_path}")

class StripConnection:
    def __init__(self, address: str, logger: SessionLogger):
        self.address = address
        self.logger = logger
        self.client: Optional[BleakClient] = None
        self.characteristic = CHARACTERISTIC_UUID

    async def connect(self):
        self.logger.event(f"{self.address}: connecting")
        self.client = BleakClient(self.address, timeout=15.0)
        await self.client.connect()
        if not self.client.is_connected:
            raise RuntimeError("BLE client reports disconnected.")
        writable = []
        for service in self.client.services:
            for char in service.characteristics:
                props = {str(item).lower() for item in char.properties}
                if "write" in props or "write-without-response" in props:
                    writable.append(char.uuid.lower())
        if self.characteristic.lower() not in writable:
            fallback = next((v for v in writable if "fff3" in v), None)
            if fallback is None:
                raise RuntimeError(f"FFF3 unavailable. Writable characteristics: {writable}")
            self.characteristic = fallback
        self.logger.event(f"{self.address}: connected using {self.characteristic}", "ble")

    async def disconnect(self):
        if self.client and self.client.is_connected:
            await self.client.disconnect()

    async def send(self, hex_command: str, step_id: str, step_label: str, command_label: str) -> bool:
        if not self.client:
            raise RuntimeError("Not connected.")
        started = time.perf_counter()
        accepted = False
        error_message = ""
        for attempt in range(2):
            try:
                await self.client.write_gatt_char(self.characteristic, bytes.fromhex(hex_command), response=False)
                accepted = True
                break
            except Exception as exc:
                error_message = str(exc)
                if attempt == 0:
                    await asyncio.sleep(RETRY_DELAY_SECONDS)
        duration_ms = (time.perf_counter() - started) * 1000.0
        self.logger.device_results.append(DeviceResult(
            timestamp=dt.datetime.now().isoformat(timespec="milliseconds"),
            step_id=step_id, step_label=step_label, command_label=command_label,
            hex_command=hex_command, address=self.address, characteristic=self.characteristic,
            accepted=accepted, duration_ms=round(duration_ms, 3), retry_used=False,
            connected_before=True, connected_after=bool(self.client.is_connected),
            error="" if accepted else error_message
        ))
        self.logger.event(f"{self.address}: {command_label} {'accepted' if accepted else 'FAILED'} ({duration_ms:.1f} ms)", "ble" if accepted else "error")
        return accepted

async def run_test():
    logger = SessionLogger()

class BleManager(QObject):
    log_signal = Signal(str, str)
    status_signal = Signal(str, str)

    def __init__(self):
        super().__init__()
        self.clients: Dict[str, BleakClient] = {}
        self.chars: Dict[str, str] = {}

    async def connect_device(self, address: str) -> bool:
        self.log_signal.emit("ble", f"Connecting to {address}...")
        self.status_signal.emit(address, "connecting")
        try:
            client = BleakClient(address, timeout=12.0)
            await client.connect()
            if not client.is_connected:
                self.status_signal.emit(address, "disconnected")
                return False

            write_char = None
            for s in client.services:
                for ch in s.characteristics:
                    props = [p.lower() for p in ch.properties]
                    if "write" in props or "write-without-response" in props:
                        if "fff3" in ch.uuid.lower() or "2022" in ch.uuid.lower():
                            write_char = ch.uuid
                            break
                        if not write_char:
                            write_char = ch.uuid

            self.clients[address] = client
            self.chars[address] = write_char or MRSTAR_WRITE_CHAR_UUID
            self.status_signal.emit(address, "connected")
            self.log_signal.emit("ble", f"Connected {address} ({self.chars[address]})")
            return True
        except Exception as e:
            self.status_signal.emit(address, "disconnected")
            self.log_signal.emit("error", f"Connection failed {address}: {e}")
            return False

    async def disconnect_device(self, address: str):
        if address in self.clients:
            try:
                await self.clients[address].disconnect()
            except Exception:
                pass
            del self.clients[address]
        self.status_signal.emit(address, "disconnected")

    async def send_hex_all(self, hex_cmd: str) -> int:
        success = 0
        for addr, client in list(self.clients.items()):
            if client.is_connected:
                ch = self.chars.get(addr, MRSTAR_WRITE_CHAR_UUID)
                try:
                    await client.write_gatt_char(ch, bytes.fromhex(hex_cmd), response=False)
                    success += 1
                except Exception as e:
                    self.log_signal.emit("error", f"Write error {addr}: {e}")
                await asyncio.sleep(0.08)
        return success

    async def pulse(self):
        await self.send_hex_all(CMD_OFF)
        await asyncio.sleep(0.18)
        await self.send_hex_all(CMD_ON)
        await asyncio.sleep(0.12)
        await self.send_hex_all(encode_mrstar_color(0, 255, 255))
        await asyncio.sleep(0.18)
        await self.send_hex_all(encode_mrstar_color(255, 0, 0))
        await self.send_hex_all(encode_mrstar_brightness(1.0))
